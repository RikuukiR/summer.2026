\subsection{比較の対象}\vspace{-0.2em}
    大前提として、比較(比較級・最上級・原級)できるのは\fitblankbf[]{形容詞}または\fitblankbf[]{副詞}である。

\subsection{比較級のつくり方}\vspace{-0.2em}
    比較級のつくり方は以下の\fitblankbf[]{3パターン}である。

    \begin{definitionbox}{\textbf{比較級のつくり方}}
        \begin{enumerate}[nosep, topsep=0pt, itemsep=0.42em]
            \item 基本は語句の語尾に\fitblankbf[]{r}、\fitblankbf[]{er}をつける
            \compareexrow{\comparecellshort{tall}{\fitblankbf[]{taller}}}{\comparecellshort{easy}{\fitblankbf[]{easier}}}
            \compareexrow{\comparecellshort{nice}{\fitblankbf[]{nicer}}}{\comparecellshort{hot}{\fitblankbf[]{hotter}}}
            \item\fitblankbf[]{長い単語}(母音が3つ以上)のときは単語の前に\fitblankbf[]{more}をつける
            \par\vspace{0.25em}
            \noindent ※\uwave{前に more をつけると比較級変化をしていることと同義なので r や er はつけない}
            \compareexrow{\comparecelllong{careful}{\fitblankbf[]{more careful}}}{\comparecelllong{beautiful}{\fitblankbf[]{more beautiful}}}
            \compareexrow{\comparecelllong{delicious}{\fitblankbf[]{more delicious}}}{\comparecelllong{interesting}{\fitblankbf[]{more interesting}}}
            \item\fitblankbf[]{不規則}に変化するもの
            \compareexrow{\comparecellja{good, well}{(\fitblankbf[]{良い}、\fitblankbf[]{上手な～})}{\fitblankbf[]{better}}}{\comparecellja{many, much}{(\fitblankbf[]{多い}、たくさんの～)}{\fitblankbf[]{more}}}
            \compareexrow{\comparecellja{bad, ill}{(\fitblankbf[]{悪い}、すぐれない～)}{\fitblankbf[]{worse}}}{\comparecellja{little}{(\fitblankbf[]{小さい}、\fitblankbf[]{少し})}{\fitblankbf[]{less}}}
        \end{enumerate}
    \end{definitionbox}
    \vspace{-0.45em}
