\subsection{原級のつくり方}\vspace{-0.35em}
    原級は \fitblankbf[]{変化} させず\fitblankbf[]{as}と\fitblankbf[]{as}で 形容詞 [ 副詞 ] をはさむだけ。

\subsection{原級を使った構文}\vspace{-0.45em}
    \begin{theorembox}{\textbf{原級の基本構文}}
        \theoremline{as + \fitblankbf[]{原級} + as + …}{「～と\fitblankbf[]{同じくらい}…」}
    \end{theorembox}
    \vspace{0.25em}

    \begin{theorembox}{\textbf{原級の否定文}}
        \theoremline{not + as + \fitblankbf[]{原級} + as + …}{「～\fitblankbf[]{ほど} … ではない」}
    \end{theorembox}
    \vspace{0.25em}

    \begin{theorembox}{\textbf{〜 times as + 原級 + as + …}}
        \theoremline{□ \fitblankbf[]{times} as + 原級 + as + …}{「～の \fitblankbf[]{□倍} …」}
    \end{theorembox}
    \vspace{0.15em}

    ※『2倍』のときは\fitblankbf[]{twice}(= two times)、『半分、$\dfrac{1}{2}$ 倍』のときは\fitblankbf[]{half}が使われることもある。

    \begin{theorembox}{\textbf{as + 原級 + as + 人 + can [ possible ]}}
        \theoremline{as + 原級 + as + 人 + (\fitblankbf[]{can}) [ \fitblankbf[]{possible} ] 〜}{「\fitblankbf[]{できるだけ} ～」}
    \end{theorembox}
    \vspace{0.2em}

    \begin{techniquebox}{\textbf{比較の否定文}}
        比較はそもそも形容詞や副詞に対して \fitblankbf[]{より深い意味} 付与する文法である。つまり、\fitblankbf[]{動詞} に対しては操作していない。
        したがって、否定文ではその文で使われる \fitblankbf[]{動詞} に着目して、否定文の構造を決定すれば良い。
    \end{techniquebox}
    \vspace{-1em}

\subsection{練習問題}\vspace{-0.55em}
    \begin{enumerate}[leftmargin=*, nosep, itemsep=1.5em]
        \item 彼は私と同じくらい速く泳げる。
        \dialogueblankpractice{He can swim as fast as me.}
        \item My brother is as tall as my father.
        \dialogueblankpractice{私の兄は父と同じくらい背が高い。}
        \item この部屋はあの部屋ほど広くない。
        \dialogueblankpractice{This room is not as large as that one.}
        \item She doesn't run as fast as her sister.
        \dialogueblankpractice{彼女は姉ほど速く走らない。}
    \end{enumerate}
