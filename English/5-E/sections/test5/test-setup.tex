% テスト類：本文はやや広め、残り余白は問題間の \vfill で均等配分
\linespread{1.1}\selectfont

% テスト用：大きな解答欄（preamble.tex の \showanswerfalse / \showanswertrue で表示切替）
% 空欄版は \phantom で解答版と同じ行数・高さを確保
\makeatletter
\newcommand{\testblank@content}[1]{%
  \ifshowanswer
    {\color{blue}\textbf{#1}}%
  \else
    \phantom{\textbf{#1}}%
  \fi
}
\newcommand{\testblank@emptyline}[1]{%
  \underline{\makebox[#1][l]{\rule{0pt}{1.55\baselineskip}}}%
}
\newcommand{\testblank@line}[2]{%
  \par\vspace{0.4em}%
  \noindent
  \def\testblank@ans{#2}%
  \ifx\testblank@ans\empty
    \testblank@emptyline{#1}%
  \else
    \underline{\makebox[#1][l]{\testblank@content{#2}}}%
  \fi
}
\newcommand{\testblank}[2][\relax]{%
  \ifx#1\relax
    \testblank@line{#2}{}%
  \else
    \testblank@line{#2}{#1}%
  \fi
}
\newcommand{\testblankinline}[2][\relax]{%
  \ifx#1\relax
    \underline{\makebox[#2][l]{\rule{0pt}{1.35\baselineskip}}}%
  \else
    \underline{\makebox[#2][l]{\testblank@content{#1}}}%
  \fi
}
\newcommand{\testnameblank}{\underline{\hspace{14em}}}
\newcommand{\testscoreblank}{\underline{\hspace{8em}}}
% 「」内に生徒が日本語を書く空欄（第2引数=幅、第1引数=解答例）
\newcommand{\testjablank}[2][\relax]{%
  \ifx#1\relax
    「\testblankinline{#2}」%
  \else
    「\testblankinline[#1]{#2}」%
  \fi
}
% 問5〜7向け：日本語空欄をやや大きく
\newcommand{\testjablankwide}[2][\relax]{%
  \par\vspace{0.08em}%
  \noindent
  \ifx#1\relax
    「\testblankinline{#2}」%
  \else
    「\underline{\makebox[#2][l]{\testblank@content{#1}}}」%
  \fi
}
% 問5〜7向け：英文解答欄をやや大きく
\newcommand{\testblankwide}[2][\relax]{%
  \par\vspace{0.2em}%
  \noindent
  \ifx#1\relax
    \testblank@line{#2}{}%
  \else
    \underline{\makebox[#2][l]{\testblank@content{#1}}}%
  \fi
}
\makeatother
