% テスト用：解答欄（\showanswerfalse / \showanswertrue で表示切替）
% 空欄版は \phantom で解答版と同じ行数・高さを確保
\makeatletter
\newcommand{\testblank@content}[1]{%
  \ifshowanswer
    {\color{blue}\textbf{#1}}%
  \else
    \phantom{\textbf{#1}}%
  \fi
}
\newcommand{\testblank@emptyline}[1]{%
  \underline{\makebox[#1][l]{\rule{0pt}{1.55\baselineskip}}}%
}
\newcommand{\testblank@line}[2]{%
  \par\vspace{0.9em}%
  \noindent
  \def\testblank@ans{#2}%
  \ifx\testblank@ans\empty
    \testblank@emptyline{#1}%
  \else
    \underline{\makebox[#1][l]{\testblank@content{#2}}}%
  \fi
}
\newcommand{\testblank}[2][\relax]{%
  \ifx#1\relax
    \testblank@line{#2}{}%
  \else
    \testblank@line{#2}{#1}%
  \fi
}
\newcommand{\testblankinline}[2][\relax]{%
  \ifx#1\relax
    \underline{\makebox[#2][l]{\rule{0pt}{1.35\baselineskip}}}%
  \else
    \underline{\makebox[#2][l]{\testblank@content{#1}}}%
  \fi
}
\newcommand{\testblankwide}[2][\relax]{%
  \par\vspace{0.9em}%
  \noindent
  \ifx#1\relax
    \testblank@line{#2}{}%
  \else
    \underline{\makebox[#2][l]{\testblank@content{#1}}}%
  \fi
}

% 問23：番号付き空欄・語数カウント記入欄
\newlength{\wordslotw}
\setlength{\wordslotw}{0.185\linewidth}
\newcommand{\compnumblank}[2][\relax]{%
  \ifx#1\relax
    \stackon[0.15ex]{\underline{\hspace{3.8em}}}{\scriptsize(#2)}%
  \else
    \stackon[0.15ex]{\underline{\makebox[3.8em][c]{\testblank@content{#1}}}}{\scriptsize(#2)}%
  \fi
}
\newcommand{\compnumblankwide}[2][\relax]{%
  \ifx#1\relax
    \stackon[0.15ex]{\underline{\hspace{0.48\linewidth}}}{\scriptsize(#2)}%
  \else
    \stackon[0.15ex]{\underline{\makebox[0.48\linewidth][l]{\testblank@content{#1}}}}{\scriptsize(#2)}%
  \fi
}
\newcommand{\wordslot}[1]{%
  \def\compos@word{#1}%
  \ifx\compos@word\empty
    \underline{\hspace{\wordslotw}}%
  \else
    \underline{\makebox[\wordslotw][c]{\testblank@content{#1}}}%
  \fi
}
\newcommand{\wordentryrowsix}[7]{%

}
