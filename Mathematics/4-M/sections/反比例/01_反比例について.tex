\subsection{反比例}

\begin{definitionbox}{\textbf{反比例}}
    \label{def:反比例}
    $y$ が $x$ の関数で, $x$ と $y$ の関係が次のような式で表されるとき, \textbf{$y$ は $x$ に反比例する}という.

    \[\fitblankbf[]{y = \frac{a}{x}} \qquad (x \neq 0)\]

    \vspace{0.5em}
    このとき, $a$ を \fitblankbf[]{比例定数} という.${}^{*}$
\end{definitionbox}

反比例は, 一方の数量が2倍, 3倍 … となると, それに伴ってもう一方の数量は, $\dfrac{1}{2}$ 倍, $\dfrac{1}{3}$ 倍 … となるような関係をいう.

% 例題2.1
\begin{example}
    $y$ が $x$ に反比例し、 $x = 4$ のとき $y = 3$ である。 $y$ を $x$ の式で表しなさい。
\end{example}

\begin{proof}\mbox{}\\
    $y = \dfrac{a}{x}$ とおく.

    \begin{hiddenanswer}
        $x = 4$, $y = 3$ を代入すると,
        \begin{align*}
            3 &= \frac{a}{4} \qquad \Longleftrightarrow \qquad a = 12
        \end{align*}
        \finalanswer{答え：$y = \dfrac{12}{x}$}
    \end{hiddenanswer}
\end{proof}

% 例題2.2
\begin{example}
    $y$ が $x$ に反比例し、 $x = 2$ のとき $y = 6$ である。 $x = -3$ のときの $y$ の値を求めなさい。
\end{example}

\begin{proof}\mbox{}\\
    $y = \dfrac{a}{x}$ とおく.

    \begin{hiddenanswer}
        $x = 2$, $y = 6$ を代入すると,
        \begin{align*}
            6 &= \frac{a}{2} \qquad \Longleftrightarrow \qquad a = 12
        \end{align*}

        $x = -3$ のときの $y$ の値を求めると,
        \begin{align*}
            y &= \frac{12}{-3} \\
            y &= -4
        \end{align*}
        \finalanswer{答え：$y = -4$}
    \end{hiddenanswer}
\end{proof}
\vspace{-2em}

\vfill
\noindent\rule{\textwidth}{0.4pt}

\vspace{0.2em}
\noindent
$^{*}$ Definition \ref{def:反比例} より, $x \neq 0$ のとき, $\bm{a = xy}$ となる.

\begin{theorembox}{\textbf{反比例のグラフ}}
    反比例：$y = \dfrac{a}{x}$ のグラフは \fitblankbf[]{双曲線} である.

    \vspace{0.25em}
    \begin{enumerate}[itemsep=0.35em, topsep=0.2em, parsep=0pt, partopsep=0pt]
        \item $a > 0$ のとき, \fitblankbf[]{右上} と \fitblankbf[]{左下} にある
        \item $a < 0$ のとき, \fitblankbf[]{右下} と \fitblankbf[]{左上} にある
        \item $x$軸 と $y$軸 には \fitblankbf[]{交わらない}
    \end{enumerate}
\end{theorembox}

% 例題2.3
\begin{example}
    次の (1)、 (2) は、 反比例の関係を表すグラフです。それぞれグラフの式を求めなさい。

    \vspace{-0.3em}
    \begin{center}
    \begin{tikzpicture}[scale=0.36, x=1cm, y=1cm]
        \tikzset{
            gline/.style={semithick},
        }

        % 格子
        \draw[step=1, gray!40, very thin] (-7,-7) grid (7,7);

        % 座標軸
        \draw[thick] (-7,0) -- (7,0) node[right] {$x$};
        \draw[thick] (0,-7) -- (0,7) node[above] {$y$};
        \node[anchor=north east] at (-0.15,-0.15) {$O$};
        \node[anchor=north] at (5,0) {$5$};
        \node[anchor=north] at (-5,0) {$-5$};
        \node[anchor=east] at (0,5) {$5$};
        \node[anchor=east] at (0,-5) {$-5$};

        % (ア) y = \frac{6}{x}
        \draw[gline] plot[domain=0.86:7, samples=120] (\x, {6/\x});
        \draw[gline] plot[domain=-7:-0.86, samples=120] (\x, {6/\x});
        \draw[dotted] (2,0) -- (2,3) -- (0,3);
        \fill (2,3) circle (1.4pt);
        \node[anchor=west] at (7,1) {{\small（1）}};

        % (イ) y = -\frac{8}{x}
        \draw[gline] plot[domain=1.15:7, samples=120] (\x, {-8/\x});
        \draw[gline] plot[domain=-7:-1.15, samples=120] (\x, {-8/\x});
        \draw[dotted] (-4,0) -- (-4,2) -- (0,2);
        \fill (-4,2) circle (1.4pt);
        \node[anchor=east] at (-7,1) {{\small（2）}};
    \end{tikzpicture}
    \end{center}
    \vspace{-0.5em}
\end{example}

\begin{proof}\mbox{}\\
    \uwave{いずれも双曲線であるから}, $y = \dfrac{a}{x}$ とおく.

    \begin{hiddenanswer}
        （1）は点 $(2,\ 3)$ を通るから
        \[a = 2 \times 3 = 6\]
        よって $y = \dfrac{6}{x}$
        \finalanswer{（1）$y = \dfrac{6}{x}$}

        （2）は点 $(-4,\ 2)$ を通るから
        \[a = (-4) \times 2 = -8\]
        よって $y = -\dfrac{8}{x}$
        \finalanswer{（2）$y = -\dfrac{8}{x}$}
    \end{hiddenanswer}
\end{proof}

% 練習問題2.1
\begin{exercise}
    反比例 $y = \dfrac{a}{x}$ （$a$ は定数）について、 $x$ の変域が $3 \leqq x \leqq 8$ であるとき、 $y$ の変域は $\dfrac{3}{2} \leqq y \leqq b$ となる。$a, b$ の値を求めなさい。
\end{exercise}

\vspace{-2.5em}
\begin{proof}\mbox{}\\
    \begin{hiddenanswer}
        \noindent
        \begin{tabular}[t]{@{}l@{\hspace{1.5em}}p{0.45\linewidth}@{}}
        \begin{minipage}[t]{0.52\linewidth}
        $x$ の変域が $3 \leqq x \leqq 8$ であるとき, $y$ の変域が $\dfrac{3}{2} \leqq y \leqq b$ となることから, 反比例 $y = \dfrac{a}{x}$ のグラフは, $x$ 座標, $y$ 座標がともに正である部分を通ることがわかる.

        よって $a > 0$

        このとき, $y = \dfrac{a}{x}$ のグラフは右の図のようになるから, グラフより
        \begin{itemize}[leftmargin=1.5em, topsep=0.2em, itemsep=0pt, parsep=0pt]
            \item $x = 3$ のとき $y = b$
            \item $x = 8$ のとき $y = \dfrac{3}{2}$
        \end{itemize}
        となることがわかる.

        $x = 8$ のとき $y = \dfrac{3}{2}$ であるから
        \[\dfrac{3}{2} = \dfrac{a}{8} \qquad \therefore\ a = 12\]

        したがって, 反比例の式は $y = \dfrac{12}{x}$

        $x = 3$ のとき $y = b$ であるから
        \[b = \dfrac{12}{3} = 4\]
        \end{minipage}
        &
        \begin{minipage}[t]{\linewidth}
        \vspace{0pt}
        \centering
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[x=1cm, y=1.8cm]
            \tikzset{
                hline/.style={very thick},
                dline/.style={semithick, dotted},
            }

            % 座標軸
            \draw[thick, ->] (0,0) -- (9.8,0) node[right] {$x$};
            \draw[thick, ->] (0,0) -- (0,5.2) node[above] {$y$};
            \node[anchor=north east] at (-0.12,-0.12) {$O$};

            % y = \frac{12}{x}（変域外は点線）
            \draw[dline] plot[domain=2.2:3.95, samples=80] (\x, {12/\x});
            \draw[dline] plot[domain=8.05:9.5, samples=80] (\x, {12/\x});

            % 変域内は太線
            \draw[hline] plot[domain=3:8, samples=80] (\x, {12/\x});

            % 補助線・点
            \draw[dotted] (3,0) -- (3,4) -- (0,4);
            \draw[dotted] (8,0) -- (8,1.5) -- (0,1.5);
            \fill (3,4) circle (1.6pt);
            \fill (8,1.5) circle (1.6pt);
            \node[anchor=north] at (3,0) {$3$};
            \node[anchor=north] at (8,0) {$8$};
            \node[anchor=east] at (0,4) {$b$};
            \node[anchor=east] at (0,1.5) {$\dfrac{3}{2}$};
        \end{tikzpicture}%
        }
        \end{minipage}
        \end{tabular}

        \alignedfinalanswer{$a = 12,\ b = 4$}
    \end{hiddenanswer}
\end{proof}

% 練習問題2.2
\begin{exercise}
    $800\ \mathrm{W}$ の出力の電子レンジで $1$ 分 $40$ 秒加熱するのが適当な食品がある。この食品を $500\ \mathrm{W}$ の出力の電子レンジで温める場合、 加熱時間を何分何秒に設定するとよいか求めなさい。ただし、 食品が温まるまでの時間は、 電子レンジの出力に反比例するものとする。
\end{exercise}

\vspace{-2.5em}
\begin{proof}\mbox{}\\
    \begin{hiddenanswer}
        $x\ \mathrm{W}$ の出力の電子レンジで温めるときの加熱時間を $y$ 秒とする.

        $y$ は $x$ に反比例するから, 比例定数を $a$ とすると $y = \dfrac{a}{x}$ と表すことができる.

        $800\ \mathrm{W}$ で $1$ 分 $40$ 秒, すなわち $100$ 秒加熱するのが適当であるから
        \begin{align*}
            100 &= \frac{a}{800} \\
            a &= 80000
        \end{align*}
        よって $y = \dfrac{80000}{x}$

        $y = \dfrac{80000}{x}$ に $x = 500$ を代入すると
        \begin{align*}
            y &= \frac{80000}{500} \\
            &= 160
        \end{align*}
        \alignedfinalanswer{$2$ 分 $40$ 秒}
    \end{hiddenanswer}
\end{proof}
