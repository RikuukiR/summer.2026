\subsection{比例}

\begin{definitionbox}{\textbf{比例}}
    \label{def:比例}
    $y$ が $x$ の関数で,$x$ と $y$ の関係が次のような式で表されるとき,\textbf{$y$ は $x$ に比例する}という.

    \[\fitblankbf[]{y = ax} \qquad (a \neq 0)\]

    \vspace{0.5em}
    このとき,$a$ を\fitblankbf[]{傾き} あるいは \fitblankbf[]{変化の割合} という.${}^{*}$
\end{definitionbox}

比例は,一方の数量が2倍,3倍 … となると,それに伴ってもう一方の数量も2倍,3倍 … となるような関係をいう.

% 例題1
\begin{example}
    $y$ が $x$ に比例し、$x = 3$ のとき $y = 6$ である。 $y$ を $x$ の式で表しなさい。
\end{example}

\begin{proof}\mbox{}\\
    $y = ax$ とおく.

    \begin{hiddenanswer}
        $x = 3$, $y = 6$ を代入すると,
        \begin{align*}
            6 = 3a \qquad \Longleftrightarrow \qquad a = 2
        \end{align*}
        \finalanswer{答え：$y = 2x$}
    \end{hiddenanswer}
\end{proof}

% 例題2
\begin{example}
    $y$ が $x$ に比例し、$x = 2$ のとき $y = -10$ である。 $x = -7$ のときの $y$ の値を求めなさい。
\end{example}

\begin{proof}\mbox{}\\
    $y = ax$ とおく.

    \begin{hiddenanswer}
        $x = 2$, $y = -10$ を代入すると,
        \begin{align*}
            -10 = 2a \qquad \Longleftrightarrow \qquad a = -5 \\
            \therefore\ y = -5x
        \end{align*}

        $x = -7$ のときの $y$ の値を求めると,
        \begin{align*}
            y &= -5 \times (-7) \\
            y &= 35
        \end{align*}
        \finalanswer{答え：$y = 35$}
    \end{hiddenanswer}
\end{proof}

\vfill
\noindent\rule{\textwidth}{0.4pt}

\vspace{0.2em}
\noindent
$^{*}$ Definition \ref{def:比例} より,$x \neq 0$ のとき,{\boldmath $a = \dfrac{y}{x}$} となる.

\subsection{比例のグラフ}

\begin{theorembox}{\textbf{比例のグラフ}}
    比例：$y = ax$ のグラフは\fitblankbf[]{原点を通る直線}である.
    \begin{enumerate}
        \item $a > 0$ のとき,グラフは\fitblankbf[]{右上がり}である.
        \item $a < 0$ のとき,グラフは\fitblankbf[]{右下がり}である.
    \end{enumerate}
\end{theorembox}

% 例題3
\begin{example}
    次の(1)〜(3)のグラフの式を、それぞれ求めなさい。

    \vspace{-0.3em}
    \begin{center}
    \begin{tikzpicture}[scale=0.36, x=1cm, y=1cm]
        \tikzset{
            gline/.style={semithick},
        }

        % 格子
        \draw[step=1, gray!40, very thin] (-7,-7) grid (7,7);

        % 座標軸
        \draw[thick] (-7,0) -- (7,0) node[right] {$x$};
        \draw[thick] (0,-7) -- (0,7) node[above] {$y$};
        \node[anchor=north east] at (-0.15,-0.15) {$O$};
        \node[anchor=north] at (5,0) {$5$};
        \node[anchor=north] at (-5,0) {$-5$};
        \node[anchor=east] at (0,5) {$5$};
        \node[anchor=east] at (0,-5) {$-5$};

        % (1) y = 3x
        \draw[gline] ({-7/3},-7) -- ({7/3},7);
        \node[anchor=west] at ({7/3+0.3},{7+0.25}) {{\small（1）}};

        % (2) y = -x
        \draw[gline] (-7,7) -- (7,-7);
        \node[anchor=north west] at (7.35,-7.35) {{\small（2）}};

        % (3) y = \frac{1}{3}x
        \draw[gline] (-7,{-7/3}) -- (7,{7/3});
        \node[anchor=west] at (7.45,{7/3+0.25}) {{\small（3）}};
    \end{tikzpicture}
    \end{center}
    \vspace{-0.5em}
\end{example}

\begin{proof}\mbox{}\\
    \uwave{いずれも原点を通る直線だから}, $y = ax$ とおく.

    \begin{hiddenanswer}
        \begin{enumerate}
            \item 右上がりの直線である. 点 $(1,\ 3)$ を通るから,
            \[3 = a \times 1 \qquad \therefore\ a = 3\]
            \finalanswer{（１）$y = 3x$}

            \item 右下がりの直線である. 点 $(1,\ -1)$ を通るから,
            \[-1 = a \times 1 \qquad \therefore\ a = -1\]
            \finalanswer{（２）$y = -x$}

            \item なだらかな右上がりの直線である. 点 $(3,\ 1)$ を通るから,
            \[1 = a \times 3 \qquad \therefore\ a = \dfrac{1}{3}\]
            \finalanswer{（３）$y = \dfrac{1}{3}x$}
        \end{enumerate}
    \end{hiddenanswer}
\end{proof}

% 練習問題1
\begin{exercise}
    比例の式 $y = -3x$ （$a \leqq x \leqq 1$）の $y$ の変域が $b \leqq y \leqq 6$ であるとき、$a, b$ の値を求めなさい。
\end{exercise}

\begin{proof}\mbox{}\\
    \begin{hiddenanswer}
        $y$ が $x$ に比例し,比例定数 $-3$ が負の数であるから $x = a$ のとき $y = 6$ となる.

        よって
        \begin{align*}
            6 &= -3a \\
            a &= -2
        \end{align*}

        また,$x = 1$ のとき $y = b$ となるから
        \begin{align*}
            b &= -3 \times 1 \\
            b &= -3
        \end{align*}
        \alignedfinalanswer{$a = -2,\ b = -3$}
    \end{hiddenanswer}
\end{proof}

% 練習問題2
\begin{exercise}
    2点 A$(3,\ 6)$、 B$(5,\ 2)$ がある。比例 $y = ax$ のグラフが AB と交わるとき、$a$ の値の範囲を求めなさい。
\end{exercise}

\begin{proof}\mbox{}\\
    \begin{hiddenanswer}
        \noindent
        \begin{tabular}[t]{@{}l@{\hspace{0.4em}}p{0.45\linewidth}@{}}
        \begin{minipage}[t]{0.52\linewidth}
        比例 $y = ax$ のグラフについて,点 A を通るときに $a$ の値がもっとも大きくなり,点 B を通るときに $a$ の値がもっとも小さくなる.
        \vspace{0.5em}

        グラフが点 A を通るとき
        \[6 = a \times 3 \qquad \therefore\ a = 2\]

        グラフが点 B を通るとき
        \[2 = a \times 5 \qquad \therefore\ a = \dfrac{2}{5}\]

        よって,$a$ の値の範囲は $\dfrac{2}{5} \leqq a \leqq 2$
        \end{minipage}
        &
        \begin{minipage}[t]{\linewidth}
        \vspace{0pt}
        \centering
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[x=1cm, y=1cm]
            \tikzset{
                pline/.style={very thick},
                abline/.style={semithick},
            }

            % 座標軸
            \draw[thick, ->] (-1.2,0) -- (7.8,0) node[right] {$x$};
            \draw[thick, ->] (0,-1.2) -- (0,8.2) node[above] {$y$};
            \node[anchor=north east] at (-0.12,-0.12) {$O$};

            % 補助線
            \draw[dotted] (3,0) -- (3,6) -- (0,6);
            \draw[dotted] (5,0) -- (5,2) -- (0,2);
            \node[anchor=north] at (3,0) {$3$};
            \node[anchor=north] at (5,0) {$5$};
            \node[anchor=east] at (0,6) {$6$};
            \node[anchor=east] at (0,2) {$2$};

            % 比例のグラフ（延長）
            \draw[pline] (-0.5,-1) -- (3.75,7.5);   % y = 2x
            \draw[pline] (-1.25,-0.5) -- (7.5,3);   % y = \frac{2}{5}x

            % 線分 AB
            \draw[abline] (3,6) -- (5,2);

            % 点 A, B
            \fill (3,6) circle (1.6pt);
            \fill (5,2) circle (1.6pt);
            \node[anchor=south east] at (2.92,6.08) {$A$};
            \node[anchor=north west] at (5.08,1.92) {$B$};
        \end{tikzpicture}%
        }
        \end{minipage}
        \end{tabular}

        \alignedfinalanswer{$\dfrac{2}{5} \leqq a \leqq 2$}
    \end{hiddenanswer}
\end{proof}

% 練習問題1.3
\begin{exercise}
    \noindent
    \begin{tabular}[t]{@{}p{0.50\linewidth}@{\hspace{1.2em}}p{0.44\linewidth}@{}}
    \vspace{0pt}
    高さが $12\ \mathrm{cm}$ の直方体の形をした容器 A、 B にそれぞれ一定の割合で水を入れる。 右のグラフは、水を入れ始めてから $x$ 分後の水面の高さを $y\ \mathrm{cm}$ とするとき、$x$ と $y$ の関係を表したものである。 次の問いに答えなさい。
    \vspace{0.5em}

    (1) 容器 B がいっぱいになるのは、容器 A がいっぱいになってから何分後か答えなさい。
    \vspace{0.35em}

    (2) 容器 B の水面の上昇速度を求めなさい。
    \vspace{0.35em}

    (3) 容器 A と容器 B の水面の高さの差が $3\ \mathrm{cm}$ になるのは、水を入れ始めてから何分後か答えなさい。
    \vspace{0.35em}

    (4) 容器 A と容器 B の水面の高さの差が $1\ \mathrm{cm}$ になるのは、水を入れ始めてから何秒後か答えなさい。 &
    \begin{minipage}[t]{\linewidth}
    \vspace{0pt}
    \centering
    \resizebox{0.85\linewidth}{!}{%
    \begin{tikzpicture}[x=1cm, y=1cm]
        \tikzset{
            gline/.style={semithick},
            ax/.style={font=\Large, inner sep=1.5pt},
            tick/.style={font=\Large, inner sep=1.5pt},
            pt/.style={font=\LARGE, inner sep=2.5pt},
        }

        % 格子（x: 0〜8, y: 0〜12）
        \draw[step=1, gray!40, very thin] (0,0) grid (8,12);

        % 座標軸
        \draw[thick, ->] (0,0) -- (8.3,0) node[ax, right, xshift=2pt] {$x$（分）};
        \draw[thick, ->] (0,0) -- (0,12.5) node[ax, above, yshift=2pt] {$y$（cm）};
        \node[tick, anchor=north east] at (-0.22,-0.22) {$O$};
        \node[tick, anchor=north] at (3,-0.18) {$3$};
        \node[tick, anchor=north] at (6,-0.18) {$6$};
        \node[tick, anchor=east] at (-0.18,5) {$5$};
        \node[tick, anchor=east] at (-0.18,10) {$10$};

        % A: y = 3x
        \draw[gline] (0,0) -- (4,12);
        \node[pt, anchor=east] at (1.85,9.1) {$A$};

        % B: y = \frac{3}{2}x
        \draw[gline] (0,0) -- (8,12);
        \node[pt, anchor=west] at (5.55,5.35) {$B$};
    \end{tikzpicture}%
    }
    \end{minipage}
    \end{tabular}
\end{exercise}

\vspace{-1.5em}
\begin{proof}\mbox{}\\
    \vspace{-1.5em}
    \begin{hiddenanswer}
        \begin{enumerate}[itemsep=0.6em, topsep=0.3em, parsep=0pt, partopsep=0pt]
            \item 容器 B のグラフについて,$y = 12$ となるのは $x = 8$ のときである. よって,容器 B がいっぱいになるのは,容器 A がいっぱいになってから $4$ 分後.
            \alignedfinalanswer{$4$ 分後}

            \item $12 \div 8 = \dfrac{3}{2}$ より,分速 $\dfrac{3}{2}\ \mathrm{cm}$
            \alignedfinalanswer{分速 $\dfrac{3}{2}\ \mathrm{cm}$}

            \item $x = 2$ のとき,容器 A の水面の高さは $6\ \mathrm{cm}$,容器 B の水面の高さは $3\ \mathrm{cm}$ であるから,容器 A と容器 B の水面の高さの差が $3\ \mathrm{cm}$ になるのは $2$ 分後.（A のグラフと B のグラフの $y$ 座標の差が $3$ になっている点を見つければよい.）
            \alignedfinalanswer{$2$ 分後}

            \item 容器 A と容器 B の水面の高さの差は一定の割合で大きくなる.（3）より,水面の高さの差が $3\ \mathrm{cm}$ になるのは $2$ 分後すなわち $120$ 秒後であるから,水面の高さの差が $1\ \mathrm{cm}$ になるのは,$120 \div 3 = 40$ より $40$ 秒後.
            \alignedfinalanswer{$40$ 秒後}
        \end{enumerate}
    \end{hiddenanswer}

    \vspace{-1em}
    \begin{altproof}
        $3 \div 1 = 3$ より,容器 A の水面の上昇速度は分速 $3\ \mathrm{cm}$

        $x$ 分後に容器 A と容器 B の水面の高さの差が $1\ \mathrm{cm}$ になるとすると
        \[3x - \dfrac{3}{2}x = 1\]
        これを解くと $x = \dfrac{2}{3}$ よって,$\dfrac{2}{3}$ 分後,すなわち $40$ 秒後に容器 A と容器 B の水面の高さの差が $1\ \mathrm{cm}$ になる.
        \alignedfinalanswer{$40$ 秒後}
    \end{altproof}
\end{proof}
